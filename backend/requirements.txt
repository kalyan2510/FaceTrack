# FaceTrack backend requirements
# Install with:  pip install -r requirements.txt
#
# Tested with Python 3.13.14. All versions below are the ones verified
# working during integration testing (Flask 3.1.3, flask-cors 6.0.5,
# numpy 2.5.4, opencv-contrib-python 4.14.0.94, openpyxl 3.1.5).
#
# opencv-contrib-python (NOT plain opencv-python) is required because it
# includes the cv2.face LBPH recognizer and Haar cascade data.
#
# Pandas is intentionally NOT required. Excel exports use OpenPyXL and CSV
# exports use Python's built-in csv module, so no pandas DLLs are loaded.
# This keeps the backend working on Windows machines where Application
# Control blocks pandas' compiled extensions.

Flask>=3.0,<4.0
flask-cors>=4.0,<7.0
Werkzeug>=3.0,<4.0
numpy>=1.26,<3.0
opencv-contrib-python>=4.10,<5.0
openpyxl>=3.1.5,<4.0

# ---------------------------------------------------------------------------
# Optional environment variables (read by backend/app.py and database.py):
#
#   FACETRACK_HOST        Host to bind (default: 127.0.0.1)
#   FACETRACK_PORT        Port to bind (default: 5000)
#   FACETRACK_DEBUG       Set "1" to enable Flask debug mode
#   FACETRACK_ADMIN_EMAIL Default admin email when DB is created (default: admin@facetrack.com)
#   FACETRACK_ADMIN_PASSWORD Default admin password when DB is created (default: Admin@123)
#
#   FACETRACK_CORS_ORIGINS  Comma-separated list of allowed CORS origins.
#                           Default: http://127.0.0.1:5500, http://localhost:5500,
#                           http://127.0.0.1:5501, http://localhost:5501
#                           Set to "*" to allow all origins (hosted deployment).
#
#   FACETRACK_SEED_DEMO    Set to "1" (or true/yes/on) to seed the database
#                           with 15 sample students and 75 days of attendance
#                           on first run. Default: off (no sample data).
#                           The admin account is always seeded when empty.
# ---------------------------------------------------------------------------
